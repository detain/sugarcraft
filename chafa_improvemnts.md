// sugar-image/src/AsyncRenderer/PacedAsyncRenderer.php (NEW)
public function renderAsync(ImageSource $image, int $w, int $h): PromiseInterface {

    $deadline = Deadline::in($this->sliceMs);              // a SLICE budget, not a request cap (E646)
